   117→/** Genera identificadores únicos (crypto.randomUUID con reserva segura). */
   118→function generateId(): string {
   119→  if (typeof crypto !== "undefined" && typeof crypto.randomUUID === "function") {
   120→    return crypto.randomUUID();
   121→  }
   122→  return `id-${Date.now()}-${Math.random().toString(36).slice(2, 10)}`;
   123→}
   124→
   125→function toISODate(date: Date): string {
   126→  const year = date.getFullYear();
   127→  const month = String(date.getMonth() + 1).padStart(2, "0");
   128→  const day = String(date.getDate()).padStart(2, "0");
   129→  return `${year}-${month}-${day}`;
   130→}
   131→
   132→/** Parsea "yyyy-MM-dd" como fecha local; null si es inválida o está vacía. */
   133→function parseDateInput(iso: string): Date | null {
   134→  if (!iso) return null;
   135→  const date = new Date(`${iso}T00:00:00`);
   136→  return Number.isNaN(date.getTime()) ? null : date;
   137→}
   138→
   139→function round2(value: number): number {
   140→  return Math.round((value + Number.EPSILON) * 100) / 100;
   141→}
   142→
   143→function numOr(value: number, fallback: number): number {
   144→  return Number.isFinite(value) ? value : fallback;
   145→}
   146→
   147→function clampNum(value: number, min: number, max: number, fallback: number): number {
   148→  if (!Number.isFinite(value)) return fallback;
   149→  return Math.min(max, Math.max(min, value));
   150→}
   151→
   152→function formatPercent(value: number): string {
   153→  return `${String(numOr(value, 0)).replace(".", ",")} %`;
   154→}

   178→function normalizeParty(raw: unknown): InvoiceParty {
   179→  const obj = (raw ?? {}) as Record<string, unknown>;
   180→  return {
   181→    name: typeof obj.name === "string" ? obj.name : "",
   182→    docId: typeof obj.docId === "string" ? obj.docId : "",
   183→    address: typeof obj.address === "string" ? obj.address : "",
   184→    email: typeof obj.email === "string" ? obj.email : "",
   185→    phone: typeof obj.phone === "string" ? obj.phone : "",
   186→  };
   187→}
   188→
   189→function normalizeLine(raw: unknown): InvoiceLine {
   190→  const obj = (raw ?? {}) as Record<string, unknown>;
   191→  return {
   192→    id: typeof obj.id === "string" && obj.id ? obj.id : generateId(),
   193→    description: typeof obj.description === "string" ? obj.description : "",
   194→    quantity: clampNum(Number(obj.quantity), 1, 1000000000, 1),
   195→    unitPrice: clampNum(Number(obj.unitPrice), 0, 1000000000000, 0),
   196→    taxRate: clampNum(Number(obj.taxRate), 0, 100, 0),
   197→  };
   198→}
   199→
   200→function normalizeConfig(raw: unknown): InvoiceConfig {
   201→  const obj = (raw ?? {}) as Record<string, unknown>;
   202→  const currency = CURRENCIES.find((c) => c.code === obj.currency)?.code ?? "COP";
   203→  return {
   204→    number: typeof obj.number === "string" && obj.number ? obj.number : "FAC-001",
   205→    issueDate:
   206→      typeof obj.issueDate === "string" && parseDateInput(obj.issueDate)
   207→        ? obj.issueDate
   208→        : toISODate(new Date()),
   209→    dueDate:
   210→      typeof obj.dueDate === "string" && parseDateInput(obj.dueDate) ? obj.dueDate : "",
   211→    currency,
   212→    taxRate: clampNum(Number(obj.taxRate), 0, 100, 0),
   213→    discountRate: clampNum(Number(obj.discountRate), 0, 100, 0),
   214→    notes: typeof obj.notes === "string" ? obj.notes : "",
   215→  };
   216→}
   217→
   218→function isValidTemplate(value: unknown): value is InvoiceTemplate {
   219→  if (!value || typeof value !== "object") return false;
   220→  const obj = value as Record<string, unknown>;
   221→  return (
   222→    typeof obj.id === "string" &&
   223→    typeof obj.nombre === "string" &&
   224→    obj.datos !== null &&
   225→    typeof obj.datos === "object"
   226→  );
   227→}
   228→
   229→/** Lee las plantillas guardadas al cargar la página (el componente es solo cliente). */
   230→function readTemplates(): InvoiceTemplate[] {
   231→  if (typeof window === "undefined") return [];
   232→  try {
   233→    const raw = window.localStorage.getItem(TEMPLATES_KEY);
   234→    if (!raw) return [];
   235→    const parsed: unknown = JSON.parse(raw);
   236→    if (!Array.isArray(parsed)) return [];
   237→    return parsed.filter(isValidTemplate);
   238→  } catch {
   239→    return [];
   240→  }
   241→}

   323→export default function InvoiceGeneratorTool() {
   324→  const [emisor, setEmisor] = useState<InvoiceParty>(EMPTY_PARTY);
   325→  const [cliente, setCliente] = useState<InvoiceParty>(EMPTY_PARTY);
   326→  const [config, setConfig] = useState<InvoiceConfig>(defaultConfig);
   327→  const [lines, setLines] = useState<InvoiceLine[]>(() => [defaultLine(0)]);
   328→
   329→  // Las plantillas se leen de localStorage al montar: el componente se carga con
   330→  // dynamic(ssr: false), por lo que el inicializador perezoso solo corre en cliente.
   331→  const [templates, setTemplates] = useState<InvoiceTemplate[]>(readTemplates);
   332→  const [selectedTemplateId, setSelectedTemplateId] = useState("");
   333→  const [saveDialogOpen, setSaveDialogOpen] = useState(false);
   334→  const [templateName, setTemplateName] = useState("");
   335→
   336→  const moneyFormatter = useMemo(
   337→    () => new Intl.NumberFormat("es", { style: "currency", currency: config.currency }),
   338→    [config.currency],
   339→  );
   340→
   341→  const formatMoney = (value: number): string => moneyFormatter.format(numOr(value, 0));
   342→
   343→  const totals = useMemo<Totals>(() => {
   344→    const items = lines.map((line) => {
   345→      const amount = round2(numOr(line.quantity, 0) * numOr(line.unitPrice, 0));
   346→      const tax = round2((amount * numOr(line.taxRate, 0)) / 100);
   347→      return { amount, tax };
   348→    });
   349→    const subtotal = round2(items.reduce((acc, it) => acc + it.amount, 0));
   350→    const discountRate = clampNum(config.discountRate, 0, 100, 0);
   351→    const discount = round2((subtotal * discountRate) / 100);
   352→    const factor = subtotal > 0 ? (subtotal - discount) / subtotal : 1;
   353→    const taxes = round2(items.reduce((acc, it) => acc + it.tax, 0) * factor);
   354→    const total = round2(subtotal - discount + taxes);
   355→    return { items, subtotal, discount, taxes, total };
   356→  }, [lines, config.discountRate]);
   357→
   358→  const updateEmisor = (patch: Partial<InvoiceParty>) =>
   359→    setEmisor((prev) => ({ ...prev, ...patch }));
   360→
   361→  const updateCliente = (patch: Partial<InvoiceParty>) =>
   362→    setCliente((prev) => ({ ...prev, ...patch }));
   363→
   364→  const updateConfig = (patch: Partial<InvoiceConfig>) =>
   365→    setConfig((prev) => ({ ...prev, ...patch }));
   366→
   367→  const updateLine = (id: string, patch: Partial<InvoiceLine>) =>
   368→    setLines((prev) =>
   369→      prev.map((line) => (line.id === id ? { ...line, ...patch } : line)),
   370→    );
   371→
   372→  const addLine = () =>
   373→    setLines((prev) => [...prev, defaultLine(config.taxRate)]);
   374→
   375→  const removeLine = (id: string) => {
   376→    setLines((prev) => {
   377→      if (prev.length <= 1) return prev;
   378→      return prev.filter((line) => line.id !== id);
   379→    });
   380→  };
   381→
   382→  /** El IVA global define el valor por defecto de las líneas nuevas y actualiza las que conservan el valor anterior. */
   383→  const handleGlobalTaxChange = (rate: number) => {
   384→    const previous = config.taxRate;
   385→    updateConfig({ taxRate: rate });
   386→    setLines((prev) =>
   387→      prev.map((line) => (line.taxRate === previous ? { ...line, taxRate: rate } : line)),
   388→    );
   389→  };
   390→
   391→  const persistTemplates = (next: InvoiceTemplate[]) => {
   392→    setTemplates(next);
   393→    try {
   394→      window.localStorage.setItem(TEMPLATES_KEY, JSON.stringify(next));
   395→    } catch {
   396→      toast.error("No se pudo guardar la plantilla en este navegador");
   397→    }
   398→  };
   399→
   400→  const saveTemplate = () => {
   401→    const name = templateName.trim();
   402→    if (!name) {
   403→      toast.error("Escribe un nombre para la plantilla");
   404→      return;
   405→    }
   406→    const existing = templates.find(
   407→      (t) => t.nombre.trim().toLowerCase() === name.toLowerCase(),
   408→    );
   409→    const template: InvoiceTemplate = {
   410→      id: existing ? existing.id : generateId(),
   411→      nombre: name,
   412→      datos: { emisor, cliente, config, lines },
   413→    };
   414→    const next = existing
   415→      ? templates.map((t) => (t.id === existing.id ? template : t))
   416→      : [...templates, template];
   417→    persistTemplates(next);
   418→    setSelectedTemplateId(template.id);
   419→    setSaveDialogOpen(false);
   420→    setTemplateName("");
   421→    toast.success(existing ? `Plantilla "${name}" actualizada` : `Plantilla "${name}" guardada`);
   422→  };
   423→
   424→  const loadTemplate = () => {
   425→    const template = templates.find((t) => t.id === selectedTemplateId);
   426→    if (!template) {
   427→      toast.error("Selecciona una plantilla para cargar");
   428→      return;
   429→    }
   430→    setEmisor(normalizeParty(template.datos.emisor));
   431→    setCliente(normalizeParty(template.datos.cliente));
   432→    setConfig(normalizeConfig(template.datos.config));
   433→    setLines(
   434→      template.datos.lines.length > 0
   435→        ? template.datos.lines.map(normalizeLine)
   436→        : [defaultLine(0)],
   437→    );
   438→    setSelectedTemplateId("");
   439→    toast.success(`Plantilla "${template.nombre}" cargada`);
   440→  };
   441→
   442→  const deleteTemplate = () => {
   443→    const template = templates.find((t) => t.id === selectedTemplateId);
   444→    if (!template) {
   445→      toast.error("Selecciona la plantilla que quieres eliminar");
   446→      return;
   447→    }
   448→    persistTemplates(templates.filter((t) => t.id !== template.id));
   449→    setSelectedTemplateId("");
   450→    toast.success(`Plantilla "${template.nombre}" eliminada`);
   451→  };
   452→
   453→  const newInvoice = () => {
   454→    setEmisor({ ...EMPTY_PARTY });
   455→    setCliente({ ...EMPTY_PARTY });
   456→    setConfig(defaultConfig());
   457→    setLines([defaultLine(0)]);
   458→    setSelectedTemplateId("");
   459→    toast.success("Nueva factura creada en blanco");
   460→  };
   461→
   462→  const validateForPdf = (): string[] => {
   463→    const problems: string[] = [];
   464→    if (!emisor.name.trim()) problems.push("el nombre del emisor");
   465→    if (!cliente.name.trim()) problems.push("el nombre del cliente");
   466→    if (!lines.some((line) => line.description.trim())) {
   467→      problems.push("al menos una línea con descripción");
   468→    }
   469→    return problems;
   470→  };
   471→
   472→  const exportPdf = () => {
   473→    const problems = validateForPdf();
   474→    if (problems.length > 0) {
   475→      toast.error(`No se puede exportar el PDF: faltan ${problems.join(", ")}.`);
   476→      return;
   477→    }
   478→    try {
   479→      buildInvoicePdf({ emisor, cliente, config, lines, totals, formatMoney });
   480→      toast.success("Factura exportada como PDF");
   481→    } catch {
   482→      toast.error("No se pudo generar el PDF. Inténtalo de nuevo.");
   483→    }
   484→  };
   485→
   486→  return (
   487→    <ToolShell>
   488→      {/* Plantillas guardadas */}
   489→      <Card>
   490→        <CardHeader>
   491→          <CardTitle className="text-base">Plantillas guardadas</CardTitle>
   492→          <CardDescription>
   493→            {templates.length === 0
   494→              ? "Aún no tienes plantillas guardadas en este navegador."
   495→              : templates.length === 1
   496→                ? "Tienes 1 plantilla guardada en este navegador."
   497→                : `Tienes ${templates.length} plantillas guardadas en este navegador.`}
   498→          </CardDescription>
   499→        </CardHeader>
   500→        <CardContent className="flex flex-col gap-3 sm:flex-row sm:items-end">
   501→          <div className="flex-1 space-y-1.5">
   502→            <Label htmlFor="invoice-template">Selecciona una plantilla</Label>
   503→            <Select
   504→              value={selectedTemplateId}
   505→              onValueChange={setSelectedTemplateId}
   506→              disabled={templates.length === 0}
   507→            >
   508→              <SelectTrigger id="invoice-template" className="w-full">
   509→                <SelectValue placeholder="Sin selección" />
   510→              </SelectTrigger>
   511→              <SelectContent>
   512→                {templates.map((template) => (
   513→                  <SelectItem key={template.id} value={template.id}>
   514→                    {template.nombre}
   515→                  </SelectItem>
   516→                ))}
   517→              </SelectContent>
   518→            </Select>
   519→          </div>
   520→          <div className="flex gap-2">
   521→            <Button
   522→              type="button"
   523→              variant="outline"
   524→              onClick={loadTemplate}
   525→              disabled={templates.length === 0 || !selectedTemplateId}
   526→            >
   527→              Cargar
   528→            </Button>
   529→            <Button
   530→              type="button"
   531→              variant="outline"
   532→              onClick={deleteTemplate}
   533→              disabled={templates.length === 0 || !selectedTemplateId}
   534→              className="text-rose-600 hover:text-rose-700"
   535→            >
   536→              <Trash2 className="size-4" aria-hidden />
   537→              Eliminar
   538→            </Button>
   539→          </div>
   540→        </CardContent>
   541→      </Card>
   542→
   543→      {/* Emisor y cliente */}
   544→      <div className="grid grid-cols-1 gap-4 lg:grid-cols-2">
   545→        <PartyCard
   546→          title="Datos del emisor"
   547→          description="Quien emite la factura. Requerido para exportar el PDF."
   548→          idPrefix="emisor"
   549→          party={emisor}
   550→          onChange={updateEmisor}
   551→          showPhone
   552→        />
   553→        <PartyCard
   554→          title="Datos del cliente"
   555→          description="Receptor de la factura. Requerido para exportar el PDF."
   556→          idPrefix="cliente"
   557→          party={cliente}
   558→          onChange={updateCliente}
   559→        />
   560→      </div>
   561→
   562→      {/* Detalles de la factura */}
   563→      <Card>
   564→        <CardHeader>
   565→          <CardTitle className="text-base">Detalles de la factura</CardTitle>
   566→          <CardDescription>
   567→            Numeración, fechas, moneda e impuestos globales. El descuento se aplica al subtotal.
   568→          </CardDescription>
   569→        </CardHeader>
   570→        <CardContent className="grid grid-cols-1 gap-4 sm:grid-cols-2">
   571→          <div className="space-y-1.5">
   572→            <Label htmlFor="inv-number">Número de factura</Label>
   573→            <Input
   574→              id="inv-number"
   575→              value={config.number}
   576→              onChange={(e) => updateConfig({ number: e.target.value })}
   577→              placeholder="FAC-001"
   578→            />
   579→          </div>
   580→          <div className="space-y-1.5">
   581→            <Label htmlFor="inv-currency">Moneda</Label>
   582→            <Select
   583→              value={config.currency}

   882→/* ---------- Generación del PDF (jsPDF + autoTable) ---------- */
   883→
   884→interface PdfPayload {
   885→  emisor: InvoiceParty;
   886→  cliente: InvoiceParty;
   887→  config: InvoiceConfig;
   888→  lines: InvoiceLine[];
   889→  totals: Totals;
   890→  formatMoney: (value: number) => string;
   891→}
   892→
   893→function buildInvoicePdf(payload: PdfPayload): void {
   894→  const { emisor, cliente, config, lines, totals, formatMoney } = payload;
   895→
   896→  const doc = new jsPDF({ orientation: "portrait", unit: "mm", format: "a4" });
   897→  const docWithTable = doc as unknown as { lastAutoTable?: { finalY: number } };
   898→
   899→  const pageWidth = doc.internal.pageSize.getWidth();
   900→  const pageHeight = doc.internal.pageSize.getHeight();
   901→  const margin = 15;
   902→  const columnWidth = (pageWidth - margin * 2 - 10) / 2;
   903→
   904→  // Encabezado
   905→  doc.setFont("helvetica", "bold");
   906→  doc.setFontSize(24);
   907→  doc.setTextColor(20, 20, 20);
   908→  doc.text("FACTURA", margin, 24);
   909→  doc.setFontSize(13);
   910→  doc.setTextColor(100, 100, 100);
   911→  doc.text(`N.º ${config.number.trim() || "FAC-001"}`, margin, 31);
   912→
   913→  doc.setFont("helvetica", "normal");
   914→  doc.setFontSize(9);
   915→  doc.setTextColor(100, 100, 100);
   916→  const rightX = pageWidth - margin;
   917→  const issue = parseDateInput(config.issueDate);
   918→  const due = parseDateInput(config.dueDate);
   919→  let headerY = 22;
   920→  doc.text(`Fecha de emisión: ${issue ? formatDate(issue) : "—"}`, rightX, headerY, {
   921→    align: "right",
   922→  });
   923→  headerY += 5;
   924→  if (due) {
   925→    doc.text(`Fecha de vencimiento: ${formatDate(due)}`, rightX, headerY, { align: "right" });
   926→    headerY += 5;
   927→  }
   928→  doc.text(`Moneda: ${config.currency}`, rightX, headerY, { align: "right" });
   929→
   930→  doc.setDrawColor(20, 20, 20);
   931→  doc.setLineWidth(0.5);
   932→  doc.line(margin, 37, pageWidth - margin, 37);
   933→
   934→  // Bloques de emisor y cliente
   935→  const drawParty = (
   936→    x: number,
   937→    title: string,
   938→    party: InvoiceParty,
   939→    includePhone: boolean,
   940→  ): number => {
   941→    doc.setFont("helvetica", "bold");
   942→    doc.setFontSize(9);
   943→    doc.setTextColor(120, 120, 120);
   944→    doc.text(title.toUpperCase(), x, 45);
   945→
   946→    doc.setFont("helvetica", "bold");
   947→    doc.setFontSize(11);
   948→    doc.setTextColor(20, 20, 20);
   949→    const nameLines = doc.splitTextToSize(party.name.trim(), columnWidth).slice(0, 2);
   950→    doc.text(nameLines, x, 51);
   951→    let y = 51 + nameLines.length * 5 + 1;
   952→
   953→    doc.setFont("helvetica", "normal");
   954→    doc.setFontSize(9);
   955→    doc.setTextColor(70, 70, 70);
   956→    const details: string[] = [];
   957→    if (party.docId.trim()) details.push(`Documento: ${party.docId.trim()}`);
   958→    if (party.address.trim()) details.push(`Dirección: ${party.address.trim()}`);
   959→    if (party.email.trim()) details.push(`Correo: ${party.email.trim()}`);
   960→    if (includePhone && party.phone.trim()) details.push(`Teléfono: ${party.phone.trim()}`);
   961→    const detailLines = details.length > 0 ? doc.splitTextToSize(details.join("\n"), columnWidth) : [];
   962→    doc.text(detailLines, x, y);
   963→    y += detailLines.length * 4.2;
   964→    return y;
   965→  };
   966→
   967→  const emisorBottom = drawParty(margin, "Emisor", emisor, true);
   968→  const clienteBottom = drawParty(margin + columnWidth + 10, "Cliente", cliente, false);
   969→
   970→  // Tabla de líneas
   971→  const tableStartY = Math.max(70, Math.max(emisorBottom, clienteBottom) + 10);
   972→  autoTable(doc, {
   973→    startY: tableStartY,
   974→    head: [["Descripción", "Cantidad", "Precio unit.", "Impuesto", "Importe"]],
   975→    body: lines.map((line, index) => [
   976→      line.description.trim(),
   977→      String(numOr(line.quantity, 0)),
   978→      formatMoney(numOr(line.unitPrice, 0)),
   979→      formatPercent(line.taxRate),
   980→      formatMoney(totals.items[index]?.amount ?? 0),
   981→    ]),
   982→    theme: "grid",
   983→    styles: {
   984→      font: "helvetica",
   985→      fontSize: 9,
   986→      cellPadding: 2.5,
   987→      lineColor: [210, 210, 210],
   988→      lineWidth: 0.15,
   989→      overflow: "linebreak",
   990→    },
   991→    headStyles: {
   992→      fillColor: [238, 238, 238],
   993→      textColor: [20, 20, 20],
   994→      fontStyle: "bold",
   995→    },
   996→    columnStyles: {
   997→      0: { cellWidth: "wrap", halign: "left" },
   998→      1: { cellWidth: 16, halign: "right" },
   999→      2: { cellWidth: 30, halign: "right" },
  1000→      3: { cellWidth: 22, halign: "right" },
  1001→      4: { cellWidth: 32, halign: "right" },
  1002→    },
  1003→    margin: { left: margin, right: margin },
  1004→  });
  1005→
  1006→  let y = docWithTable.lastAutoTable?.finalY ?? tableStartY;
  1007→  if (y > pageHeight - 70) {
  1008→    doc.addPage();
  1009→    y = margin;
  1010→  }
  1011→
  1012→  // Bloque de totales alineado a la derecha
  1013→  y += 8;
  1014→  const labelX = pageWidth - margin - 62;
  1015→  const valueX = pageWidth - margin;
  1016→  const drawTotalRow = (label: string, value: string, bold = false, size = 9.5) => {
  1017→    doc.setFont("helvetica", bold ? "bold" : "normal");
  1018→    doc.setFontSize(size);
  1019→    doc.setTextColor(bold ? 20 : 70, bold ? 20 : 70, bold ? 20 : 70);
  1020→    doc.text(label, labelX, y);
  1021→    doc.text(value, valueX, y, { align: "right" });
  1022→    y += size > 10 ? 8 : 5.6;
  1023→  };
  1024→
  1025→  drawTotalRow("Subtotal", formatMoney(totals.subtotal));
  1026→  if (totals.discount > 0) {
  1027→    drawTotalRow(
  1028→      `Descuento (${formatPercent(config.discountRate)})`,
  1029→      `- ${formatMoney(totals.discount)}`,
  1030→    );
  1031→  }
  1032→  drawTotalRow("Total impuestos", formatMoney(totals.taxes));
  1033→  y += 1;
  1034→  doc.setDrawColor(20, 20, 20);
  1035→  doc.setLineWidth(0.3);
  1036→  doc.line(labelX, y, valueX, y);
  1037→  y += 7;
  1038→  drawTotalRow("TOTAL", formatMoney(totals.total), true, 13);
  1039→
  1040→  // Notas
  1041→  if (config.notes.trim()) {
  1042→    if (y > pageHeight - 45) {
  1043→      doc.addPage();
  1044→      y = margin;
  1045→    }
  1046→    y += 10;
  1047→    doc.setFont("helvetica", "bold");
  1048→    doc.setFontSize(9);
  1049→    doc.setTextColor(120, 120, 120);
  1050→    doc.text("NOTAS", margin, y);
  1051→    doc.setFont("helvetica", "normal");
  1052→    doc.setFontSize(9.5);
  1053→    doc.setTextColor(40, 40, 40);
  1054→    const noteLines = doc.splitTextToSize(config.notes.trim(), pageWidth - margin * 2);
  1055→    doc.text(noteLines, margin, y + 5);
  1056→    y += 5 + noteLines.length * 4.2;
  1057→  }
  1058→
  1059→  // Pie de página en todas las páginas
  1060→  const pageCount = doc.getNumberOfPages();
  1061→  for (let page = 1; page <= pageCount; page += 1) {
  1062→    doc.setPage(page);
  1063→    doc.setFont("helvetica", "normal");
  1064→    doc.setFontSize(8);
  1065→    doc.setTextColor(130, 130, 130);
  1066→    doc.text("Generado con OSK APPS — Herramientas online gratis", margin, pageHeight - 8);
  1067→    doc.text(
  1068→      `Fecha de generación: ${formatDate(new Date())} · Página ${page} de ${pageCount}`,
  1069→      rightX,
  1070→      pageHeight - 8,
  1071→      { align: "right" },
  1072→    );
  1073→  }
  1074→
  1075→  const safeNumber =
  1076→    (config.number.trim() || "FAC-001").replace(/[\\/:*?"<>|\s]+/g, "-") || "FAC-001";
  1077→  doc.save(`factura-${safeNumber}.pdf`);
  1078→}
  1079→
