   114→/** Genera identificadores únicos (crypto.randomUUID con reserva segura). */
   115→function generateId(): string {
   116→  if (typeof crypto !== "undefined" && typeof crypto.randomUUID === "function") {
   117→    return crypto.randomUUID();
   118→  }
   119→  return `id-${Date.now()}-${Math.random().toString(36).slice(2, 10)}`;
   120→}
   121→
   122→function toISODate(date: Date): string {
   123→  const year = date.getFullYear();
   124→  const month = String(date.getMonth() + 1).padStart(2, "0");
   125→  const day = String(date.getDate()).padStart(2, "0");
   126→  return `${year}-${month}-${day}`;
   127→}
   128→
   129→/** Parsea "yyyy-MM-dd" como fecha local; null si es inválida o está vacía. */
   130→function parseDateInput(iso: string): Date | null {
   131→  if (!iso) return null;
   132→  const date = new Date(`${iso}T00:00:00`);
   133→  return Number.isNaN(date.getTime()) ? null : date;
   134→}
   135→
   136→function round2(value: number): number {
   137→  return Math.round((value + Number.EPSILON) * 100) / 100;
   138→}
   139→
   140→function numOr(value: number, fallback: number): number {
   141→  return Number.isFinite(value) ? value : fallback;
   142→}
   143→
   144→function clampNum(value: number, min: number, max: number, fallback: number): number {
   145→  if (!Number.isFinite(value)) return fallback;
   146→  return Math.min(max, Math.max(min, value));
   147→}
   148→
   149→function formatPercent(value: number): string {
   150→  return `${String(numOr(value, 0)).replace(".", ",")} %`;
   151→}

   175→function normalizeParty(raw: unknown): InvoiceParty {
   176→  const obj = (raw ?? {}) as Record<string, unknown>;
   177→  return {
   178→    name: typeof obj.name === "string" ? obj.name : "",
   179→    docId: typeof obj.docId === "string" ? obj.docId : "",
   180→    address: typeof obj.address === "string" ? obj.address : "",
   181→    email: typeof obj.email === "string" ? obj.email : "",
   182→    phone: typeof obj.phone === "string" ? obj.phone : "",
   183→  };
   184→}
   185→
   186→function normalizeLine(raw: unknown): InvoiceLine {
   187→  const obj = (raw ?? {}) as Record<string, unknown>;
   188→  return {
   189→    id: typeof obj.id === "string" && obj.id ? obj.id : generateId(),
   190→    description: typeof obj.description === "string" ? obj.description : "",
   191→    quantity: clampNum(Number(obj.quantity), 1, 1000000000, 1),
   192→    unitPrice: clampNum(Number(obj.unitPrice), 0, 1000000000000, 0),
   193→    taxRate: clampNum(Number(obj.taxRate), 0, 100, 0),
   194→  };
   195→}
   196→
   197→function normalizeConfig(raw: unknown): InvoiceConfig {
   198→  const obj = (raw ?? {}) as Record<string, unknown>;
   199→  const currency = CURRENCIES.find((c) => c.code === obj.currency)?.code ?? "COP";
   200→  return {
   201→    number: typeof obj.number === "string" && obj.number ? obj.number : "FAC-001",
   202→    issueDate:
   203→      typeof obj.issueDate === "string" && parseDateInput(obj.issueDate)
   204→        ? obj.issueDate
   205→        : toISODate(new Date()),
   206→    dueDate:
   207→      typeof obj.dueDate === "string" && parseDateInput(obj.dueDate) ? obj.dueDate : "",
   208→    currency,
   209→    taxRate: clampNum(Number(obj.taxRate), 0, 100, 0),
   210→    discountRate: clampNum(Number(obj.discountRate), 0, 100, 0),
   211→    notes: typeof obj.notes === "string" ? obj.notes : "",
   212→  };
   213→}
   214→
   215→function isValidTemplate(value: unknown): value is InvoiceTemplate {
   216→  if (!value || typeof value !== "object") return false;
   217→  const obj = value as Record<string, unknown>;
   218→  return (
   219→    typeof obj.id === "string" &&
   220→    typeof obj.name === "string" &&
   221→    Array.isArray(obj.lines) &&
   222→    typeof obj.emisor === "object" &&
   223→    typeof obj.cliente === "object" &&
   224→    typeof obj.config === "object"
   225→  );
   226→}
   227→
   228→/** Lee las plantillas guardadas al cargar la página (el componente es solo cliente). */
   229→function readTemplates(): InvoiceTemplate[] {
   230→  if (typeof window === "undefined") return [];
   231→  try {
   232→    const raw = window.localStorage.getItem(TEMPLATES_KEY);
   233→    if (!raw) return [];
   234→    const parsed: unknown = JSON.parse(raw);
   235→    if (!Array.isArray(parsed)) return [];
   236→    return parsed.filter(isValidTemplate);
   237→  } catch {
   238→    return [];
   239→  }
   240→}

   322→export default function InvoiceGeneratorTool() {
   323→  const [emisor, setEmisor] = useState<InvoiceParty>(EMPTY_PARTY);
   324→  const [cliente, setCliente] = useState<InvoiceParty>(EMPTY_PARTY);
   325→  const [config, setConfig] = useState<InvoiceConfig>(defaultConfig);
   326→  const [lines, setLines] = useState<InvoiceLine[]>(() => [defaultLine(0)]);
   327→
   328→  const [templates, setTemplates] = useState<InvoiceTemplate[]>(readTemplates);
   329→  const [selectedTemplateId, setSelectedTemplateId] = useState("");
   330→  const [saveDialogOpen, setSaveDialogOpen] = useState(false);
   331→  const [templateName, setTemplateName] = useState("");
   332→
   333→  // Carga inicial de plantillas desde localStorage (ver readTemplates)
   334→  const moneyFormatter = useMemo(
   335→    () => new Intl.NumberFormat("es", { style: "currency", currency: config.currency }),
   336→    [config.currency],
   337→  );
   338→
   339→  const formatMoney = (value: number): string => moneyFormatter.format(numOr(value, 0));
   340→
   341→  const totals = useMemo<Totals>(() => {
   342→    const items = lines.map((line) => {
   343→      const amount = round2(numOr(line.quantity, 0) * numOr(line.unitPrice, 0));
   344→      const tax = round2((amount * numOr(line.taxRate, 0)) / 100);
   345→      return { amount, tax };
   346→    });
   347→    const subtotal = round2(items.reduce((acc, it) => acc + it.amount, 0));
   348→    const discountRate = clampNum(config.discountRate, 0, 100, 0);
   349→    const discount = round2((subtotal * discountRate) / 100);
   350→    const factor = subtotal > 0 ? (subtotal - discount) / subtotal : 1;
   351→    const taxes = round2(items.reduce((acc, it) => acc + it.tax, 0) * factor);
   352→    const total = round2(subtotal - discount + taxes);
   353→    return { items, subtotal, discount, taxes, total };
   354→  }, [lines, config.discountRate]);
   355→
   356→  const updateEmisor = (patch: Partial<InvoiceParty>) =>
   357→    setEmisor((prev) => ({ ...prev, ...patch }));
   358→
   359→  const updateCliente = (patch: Partial<InvoiceParty>) =>
   360→    setCliente((prev) => ({ ...prev, ...patch }));
   361→
   362→  const updateConfig = (patch: Partial<InvoiceConfig>) =>
   363→    setConfig((prev) => ({ ...prev, ...patch }));
   364→
   365→  const updateLine = (id: string, patch: Partial<InvoiceLine>) =>
   366→    setLines((prev) =>
   367→      prev.map((line) => (line.id === id ? { ...line, ...patch } : line)),
   368→    );
   369→
   370→  const addLine = () =>
   371→    setLines((prev) => [...prev, defaultLine(config.taxRate)]);
   372→
   373→  const removeLine = (id: string) => {
   374→    setLines((prev) => {
   375→      if (prev.length <= 1) return prev;
   376→      return prev.filter((line) => line.id !== id);
   377→    });
   378→  };
   379→
   380→  /** El IVA global define el valor por defecto de las líneas nuevas y actualiza las que conservan el valor anterior. */
   381→  const handleGlobalTaxChange = (rate: number) => {
   382→    const previous = config.taxRate;
   383→    updateConfig({ taxRate: rate });
   384→    setLines((prev) =>
   385→      prev.map((line) => (line.taxRate === previous ? { ...line, taxRate: rate } : line)),
   386→    );
   387→  };
   388→
   389→  const persistTemplates = (next: InvoiceTemplate[]) => {
   390→    setTemplates(next);
   391→    try {
   392→      window.localStorage.setItem(TEMPLATES_KEY, JSON.stringify(next));
   393→    } catch {
   394→      toast.error("No se pudo guardar la plantilla en este navegador");
   395→    }
   396→  };
   397→
   398→  const saveTemplate = () => {
   399→    const name = templateName.trim();
   400→    if (!name) {
   401→      toast.error("Escribe un nombre para la plantilla");
   402→      return;
   403→    }
   404→    const existing = templates.find(
   405→      (t) => t.name.trim().toLowerCase() === name.toLowerCase(),
   406→    );
   407→    const template: InvoiceTemplate = {
   408→      id: existing ? existing.id : generateId(),
   409→      name,
   410→      savedAt: new Date().toISOString(),
   411→      emisor,
   412→      cliente,
   413→      config,
   414→      lines,
   415→    };
   416→    const next = existing
   417→      ? templates.map((t) => (t.id === existing.id ? template : t))
   418→      : [...templates, template];
   419→    persistTemplates(next);
   420→    setSelectedTemplateId(template.id);
   421→    setSaveDialogOpen(false);
   422→    setTemplateName("");
   423→    toast.success(existing ? `Plantilla "${name}" actualizada` : `Plantilla "${name}" guardada`);
   424→  };
   425→
   426→  const loadTemplate = () => {
   427→    const template = templates.find((t) => t.id === selectedTemplateId);
   428→    if (!template) {
   429→      toast.error("Selecciona una plantilla para cargar");
   430→      return;
   431→    }
   432→    setEmisor(normalizeParty(template.emisor));
   433→    setCliente(normalizeParty(template.cliente));
   434→    setConfig(normalizeConfig(template.config));
   435→    setLines(
   436→      template.lines.length > 0 ? template.lines.map(normalizeLine) : [defaultLine(0)],
   437→    );
   438→    setSelectedTemplateId("");
   439→    toast.success(`Plantilla "${template.name}" cargada`);
   440→  };
   441→
   442→  const deleteTemplate = () => {
   443→    const template = templates.find((t) => t.id === selectedTemplateId);
   444→    if (!template) {
   445→      toast.error("Selecciona la plantilla que quieres eliminar");
   446→      return;
   447→    }
   448→    persistTemplates(templates.filter((t) => t.id !== template.id));
   449→    setSelectedTemplateId("");
   450→    toast.success(`Plantilla "${template.name}" eliminada`);
   451→  };
   452→
   453→  const resetInvoice = () => {
   454→    setEmisor({ ...EMPTY_PARTY });
   455→    setCliente({ ...EMPTY_PARTY });
   456→    setConfig(defaultConfig());
   457→    setLines([defaultLine(0)]);
   458→    setSelectedTemplateId("");
   459→    toast.success("Se creó una factura en blanco");
   460→  };
   461→
   462→  const newInvoice = () => {
   463→    toast("Se creará una factura en blanco. Los datos no guardados se perderán.", {
   464→      action: { label: "Confirmar", onClick: resetInvoice },
   465→      duration: 8000,
   466→    });
   467→  };
   468→
   469→  const validateForPdf = (): string[] => {
   470→    const problems: string[] = [];
   471→    if (!emisor.name.trim()) problems.push("el nombre del emisor");
   472→    if (!cliente.name.trim()) problems.push("el nombre del cliente");
   473→    if (!lines.some((line) => line.description.trim())) {
   474→      problems.push("al menos una línea con descripción");
   475→    }
   476→    return problems;
   477→  };
   478→
   479→  const exportPdf = () => {
   480→    const problems = validateForPdf();
   481→    if (problems.length > 0) {
   482→      toast.error(`No se puede exportar el PDF: faltan ${problems.join(", ")}.`);
   483→      return;
   484→    }
   485→    try {
   486→      buildInvoicePdf({ emisor, cliente, config, lines, totals, formatMoney });
   487→      toast.success("Factura exportada como PDF");
   488→    } catch {
   489→      toast.error("No se pudo generar el PDF. Inténtalo de nuevo.");
   490→    }
   491→  };
   492→
   493→  return (
   494→    <ToolShell>
   495→      {/* Plantillas guardadas */}
   496→      <Card>
   497→        <CardHeader>
   498→          <CardTitle className="text-base">Plantillas guardadas</CardTitle>
   499→          <CardDescription>
   500→            {templates.length === 0
   501→              ? "Aún no tienes plantillas guardadas en este navegador."
   502→              : templates.length === 1
   503→                ? "Tienes 1 plantilla guardada en este navegador."
   504→                : `Tienes ${templates.length} plantillas guardadas en este navegador.`}
   505→          </CardDescription>
   506→        </CardHeader>
   507→        <CardContent className="flex flex-col gap-3 sm:flex-row sm:items-end">
   508→          <div className="flex-1 space-y-1.5">
   509→            <Label htmlFor="invoice-template">Selecciona una plantilla</Label>
   510→            <Select
   511→              value={selectedTemplateId}
   512→              onValueChange={setSelectedTemplateId}
   513→              disabled={templates.length === 0}
   514→            >
   515→              <SelectTrigger id="invoice-template" className="w-full">
   516→                <SelectValue placeholder="Sin selección" />
   517→              </SelectTrigger>
   518→              <SelectContent>
   519→                {templates.map((template) => (
   520→                  <SelectItem key={template.id} value={template.id}>
   521→                    {template.name} ({formatDate(template.savedAt)})
   522→                  </SelectItem>
   523→                ))}
   524→              </SelectContent>
   525→            </Select>
   526→          </div>
   527→          <div className="flex gap-2">
   528→            <Button
   529→              type="button"
   530→              variant="outline"
   531→              onClick={loadTemplate}
   532→              disabled={templates.length === 0 || !selectedTemplateId}
   533→            >
   534→              Cargar
   535→            </Button>
   536→            <Button
   537→              type="button"
   538→              variant="outline"
   539→              onClick={deleteTemplate}
   540→              disabled={templates.length === 0 || !selectedTemplateId}
   541→              className="text-rose-600 hover:text-rose-700"
   542→            >
   543→              <Trash2 className="size-4" aria-hidden />
   544→              Eliminar
   545→            </Button>
   546→          </div>
   547→        </CardContent>
   548→      </Card>
   549→
   550→      {/* Emisor y cliente */}
   551→      <div className="grid grid-cols-1 gap-4 lg:grid-cols-2">
   552→        <PartyCard
   553→          title="Datos del emisor"
   554→          description="Quien emite la factura. Requerido para exportar el PDF."
   555→          idPrefix="emisor"
   556→          party={emisor}
   557→          onChange={updateEmisor}
   558→          showPhone
   559→        />
   560→        <PartyCard
   561→          title="Datos del cliente"
   562→          description="Receptor de la factura. Requerido para exportar el PDF."
   563→          idPrefix="cliente"
   564→          party={cliente}
   565→          onChange={updateCliente}
   566→        />
   567→      </div>
   568→
   569→      {/* Detalles de la factura */}
   570→      <Card>
   571→        <CardHeader>
   572→          <CardTitle className="text-base">Detalles de la factura</CardTitle>
   573→          <CardDescription>
   574→            Numeración, fechas, moneda e impuestos globales. El descuento se aplica al subtotal.
   575→          </CardDescription>
   576→        </CardHeader>
   577→        <CardContent className="grid grid-cols-1 gap-4 sm:grid-cols-2">
   578→          <div className="space-y-1.5">
   579→            <Label htmlFor="inv-number">Número de factura</Label>
   580→            <Input
   581→              id="inv-number"
   582→              value={config.number}
   583→              onChange={(e) => updateConfig({ number: e.target.value })}
   584→              placeholder="FAC-001"
   585→            />

   891→/* ---------- Generación del PDF (jsPDF + autoTable) ---------- */
   892→
   893→interface PdfPayload {
   894→  emisor: InvoiceParty;
   895→  cliente: InvoiceParty;
   896→  config: InvoiceConfig;
   897→  lines: InvoiceLine[];
   898→  totals: Totals;
   899→  formatMoney: (value: number) => string;
   900→}
   901→
   902→function buildInvoicePdf(payload: PdfPayload): void {
   903→  const { emisor, cliente, config, lines, totals, formatMoney } = payload;
   904→
   905→  const doc = new jsPDF({ orientation: "portrait", unit: "mm", format: "a4" });
   906→  const docWithTable = doc as unknown as { lastAutoTable?: { finalY: number } };
   907→
   908→  const pageWidth = doc.internal.pageSize.getWidth();
   909→  const pageHeight = doc.internal.pageSize.getHeight();
   910→  const margin = 15;
   911→  const columnWidth = (pageWidth - margin * 2 - 10) / 2;
   912→
   913→  // Encabezado
   914→  doc.setFont("helvetica", "bold");
   915→  doc.setFontSize(24);
   916→  doc.setTextColor(20, 20, 20);
   917→  doc.text("FACTURA", margin, 24);
   918→  doc.setFontSize(13);
   919→  doc.setTextColor(100, 100, 100);
   920→  doc.text(`N.º ${config.number.trim() || "FAC-001"}`, margin, 31);
   921→
   922→  doc.setFont("helvetica", "normal");
   923→  doc.setFontSize(9);
   924→  doc.setTextColor(100, 100, 100);
   925→  const rightX = pageWidth - margin;
   926→  const issue = parseDateInput(config.issueDate);
   927→  const due = parseDateInput(config.dueDate);
   928→  let headerY = 22;
   929→  doc.text(`Fecha de emisión: ${issue ? formatDate(issue) : "—"}`, rightX, headerY, {
   930→    align: "right",
   931→  });
   932→  headerY += 5;
   933→  if (due) {
   934→    doc.text(`Fecha de vencimiento: ${formatDate(due)}`, rightX, headerY, { align: "right" });
   935→    headerY += 5;
   936→  }
   937→  doc.text(`Moneda: ${config.currency}`, rightX, headerY, { align: "right" });
   938→
   939→  doc.setDrawColor(20, 20, 20);
   940→  doc.setLineWidth(0.5);
   941→  doc.line(margin, 37, pageWidth - margin, 37);
   942→
   943→  // Bloques de emisor y cliente
   944→  const drawParty = (
   945→    x: number,
   946→    title: string,
   947→    party: InvoiceParty,
   948→    includePhone: boolean,
   949→  ): number => {
   950→    doc.setFont("helvetica", "bold");
   951→    doc.setFontSize(9);
   952→    doc.setTextColor(120, 120, 120);
   953→    doc.text(title.toUpperCase(), x, 45);
   954→
   955→    doc.setFont("helvetica", "bold");
   956→    doc.setFontSize(11);
   957→    doc.setTextColor(20, 20, 20);
   958→    const nameLines = doc.splitTextToSize(party.name.trim(), columnWidth).slice(0, 2);
   959→    doc.text(nameLines, x, 51);
   960→    let y = 51 + nameLines.length * 5 + 1;
   961→
   962→    doc.setFont("helvetica", "normal");
   963→    doc.setFontSize(9);
   964→    doc.setTextColor(70, 70, 70);
   965→    const details: string[] = [];
   966→    if (party.docId.trim()) details.push(`Documento: ${party.docId.trim()}`);
   967→    if (party.address.trim()) details.push(`Dirección: ${party.address.trim()}`);
   968→    if (party.email.trim()) details.push(`Correo: ${party.email.trim()}`);
   969→    if (includePhone && party.phone.trim()) details.push(`Teléfono: ${party.phone.trim()}`);
   970→    const detailLines = details.length > 0 ? doc.splitTextToSize(details.join("\n"), columnWidth) : [];
   971→    doc.text(detailLines, x, y);
   972→    y += detailLines.length * 4.2;
   973→    return y;
   974→  };
   975→
   976→  const emisorBottom = drawParty(margin, "Emisor", emisor, true);
   977→  const clienteBottom = drawParty(margin + columnWidth + 10, "Cliente", cliente, false);
   978→
   979→  // Tabla de líneas
   980→  const tableStartY = Math.max(70, Math.max(emisorBottom, clienteBottom) + 10);
   981→  autoTable(doc, {
   982→    startY: tableStartY,
   983→    head: [["Descripción", "Cant.", "Precio unit.", "Impuesto", "Importe"]],
   984→    body: lines.map((line, index) => [
   985→      line.description.trim(),
   986→      String(numOr(line.quantity, 0)),
   987→      formatMoney(numOr(line.unitPrice, 0)),
   988→      formatPercent(line.taxRate),
   989→      formatMoney(totals.items[index]?.amount ?? 0),
   990→    ]),
   991→    theme: "grid",
   992→    styles: {
   993→      font: "helvetica",
   994→      fontSize: 9,
   995→      cellPadding: 2.5,
   996→      lineColor: [210, 210, 210],
   997→      lineWidth: 0.15,
   998→      overflow: "linebreak",
   999→    },
  1000→    headStyles: {
  1001→      fillColor: [238, 238, 238],
  1002→      textColor: [20, 20, 20],
  1003→      fontStyle: "bold",
  1004→    },
  1005→    columnStyles: {
  1006→      0: { cellWidth: "wrap", halign: "left" },
  1007→      1: { cellWidth: 16, halign: "right" },
  1008→      2: { cellWidth: 30, halign: "right" },
  1009→      3: { cellWidth: 22, halign: "right" },
  1010→      4: { cellWidth: 32, halign: "right" },
  1011→    },
  1012→    margin: { left: margin, right: margin },
  1013→  });
  1014→
  1015→  let y = docWithTable.lastAutoTable?.finalY ?? tableStartY;
  1016→  if (y > pageHeight - 70) {
  1017→    doc.addPage();
  1018→    y = margin;
  1019→  }
  1020→
  1021→  // Bloque de totales alineado a la derecha
  1022→  y += 8;
  1023→  const labelX = pageWidth - margin - 62;
  1024→  const valueX = pageWidth - margin;
  1025→  const drawTotalRow = (label: string, value: string, bold = false, size = 9.5) => {
  1026→    doc.setFont("helvetica", bold ? "bold" : "normal");
  1027→    doc.setFontSize(size);
  1028→    doc.setTextColor(bold ? 20 : 70, bold ? 20 : 70, bold ? 20 : 70);
  1029→    doc.text(label, labelX, y);
  1030→    doc.text(value, valueX, y, { align: "right" });
  1031→    y += size > 10 ? 8 : 5.6;
  1032→  };
  1033→
  1034→  drawTotalRow("Subtotal", formatMoney(totals.subtotal));
  1035→  if (totals.discount > 0) {
  1036→    drawTotalRow(
  1037→      `Descuento (${formatPercent(config.discountRate)})`,
  1038→      `- ${formatMoney(totals.discount)}`,
  1039→    );
  1040→  }
  1041→  drawTotalRow("Total impuestos", formatMoney(totals.taxes));
  1042→  y += 1;
  1043→  doc.setDrawColor(20, 20, 20);
  1044→  doc.setLineWidth(0.3);
  1045→  doc.line(labelX, y, valueX, y);
  1046→  y += 7;
  1047→  drawTotalRow("TOTAL", formatMoney(totals.total), true, 13);
  1048→
  1049→  // Notas
  1050→  if (config.notes.trim()) {
  1051→    if (y > pageHeight - 45) {
  1052→      doc.addPage();
  1053→      y = margin;
  1054→    }
  1055→    y += 10;
  1056→    doc.setFont("helvetica", "bold");
  1057→    doc.setFontSize(9);
  1058→    doc.setTextColor(120, 120, 120);
  1059→    doc.text("NOTAS", margin, y);
  1060→    doc.setFont("helvetica", "normal");
  1061→    doc.setFontSize(9.5);
  1062→    doc.setTextColor(40, 40, 40);
  1063→    const noteLines = doc.splitTextToSize(config.notes.trim(), pageWidth - margin * 2);
  1064→    doc.text(noteLines, margin, y + 5);
  1065→    y += 5 + noteLines.length * 4.2;
  1066→  }
  1067→
  1068→  // Pie de página en todas las páginas
  1069→  const pageCount = doc.getNumberOfPages();
  1070→  for (let page = 1; page <= pageCount; page += 1) {
  1071→    doc.setPage(page);
  1072→    doc.setFont("helvetica", "normal");
  1073→    doc.setFontSize(8);
  1074→    doc.setTextColor(130, 130, 130);
  1075→    doc.text("Generado con OSK APPS — Herramientas online gratis", margin, pageHeight - 8);
  1076→    doc.text(
  1077→      `Fecha de generación: ${formatDate(new Date())} · Página ${page} de ${pageCount}`,
  1078→      rightX,
  1079→      pageHeight - 8,
  1080→      { align: "right" },
  1081→    );
  1082→  }
  1083→
  1084→  const safeNumber =
  1085→    (config.number.trim() || "FAC-001").replace(/[\\/:*?"<>|\s]+/g, "-") || "FAC-001";
  1086→  doc.save(`factura-${safeNumber}.pdf`);
  1087→}
  1088→
