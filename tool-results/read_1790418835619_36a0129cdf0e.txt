   114→   114→/** Genera identificadores únicos (crypto.randomUUID con reserva segura). */
   115→   115→function generateId(): string {
   116→   116→  if (typeof crypto !== "undefined" && typeof crypto.randomUUID === "function") {
   117→   117→    return crypto.randomUUID();
   118→   118→  }
   119→   119→  return `id-${Date.now()}-${Math.random().toString(36).slice(2, 10)}`;
   120→   120→}

   122→   122→function toISODate(date: Date): string {
   123→   123→  const year = date.getFullYear();
   124→   124→  const month = String(date.getMonth() + 1).padStart(2, "0");
   125→   125→  const day = String(date.getDate()).padStart(2, "0");
   126→   126→  return `${year}-${month}-${day}`;
   127→   127→}

   129→   129→/** Parsea "yyyy-MM-dd" como fecha local; null si es inválida o está vacía. */
   130→   130→function parseDateInput(iso: string): Date | null {
   131→   131→  if (!iso) return null;
   132→   132→  const date = new Date(`${iso}T00:00:00`);
   133→   133→  return Number.isNaN(date.getTime()) ? null : date;
   134→   134→}

   136→   136→function round2(value: number): number {
   137→   137→  return Math.round((value + Number.EPSILON) * 100) / 100;
   138→   138→}
   139→   139→
   140→   140→function numOr(value: number, fallback: number): number {
   141→   141→  return Number.isFinite(value) ? value : fallback;
   142→   142→}
   143→   143→
   144→   144→function clampNum(value: number, min: number, max: number, fallback: number): number {
   145→   145→  if (!Number.isFinite(value)) return fallback;
   146→   146→  return Math.min(max, Math.max(min, value));
   147→   147→}
   148→   148→
   149→   149→function formatPercent(value: number): string {
   150→   150→  return `${String(numOr(value, 0)).replace(".", ",")} %`;

   186→   186→function normalizeLine(raw: unknown): InvoiceLine {
   187→   187→  const obj = (raw ?? {}) as Record<string, unknown>;
   188→   188→  return {
   189→   189→    id: typeof obj.id === "string" && obj.id ? obj.id : generateId(),
   190→   190→    description: typeof obj.description === "string" ? obj.description : "",
   191→   191→    quantity: clampNum(Number(obj.quantity), 1, 1000000000, 1),
   192→   192→    unitPrice: clampNum(Number(obj.unitPrice), 0, 1000000000000, 0),
   193→   193→    taxRate: clampNum(Number(obj.taxRate), 0, 100, 0),
   194→   194→  };
   195→   195→}

   197→   197→function normalizeConfig(raw: unknown): InvoiceConfig {
   198→   198→  const obj = (raw ?? {}) as Record<string, unknown>;
   199→   199→  const currency = CURRENCIES.find((c) => c.code === obj.currency)?.code ?? "COP";
   200→   200→  return {
   201→   201→    number: typeof obj.number === "string" && obj.number ? obj.number : "FAC-001",
   202→   202→    issueDate:
   203→   203→      typeof obj.issueDate === "string" && parseDateInput(obj.issueDate)
   204→   204→        ? obj.issueDate
   205→   205→        : toISODate(new Date()),
   206→   206→    dueDate:
   207→   207→      typeof obj.dueDate === "string" && parseDateInput(obj.dueDate) ? obj.dueDate : "",
   208→   208→    currency,
   209→   209→    taxRate: clampNum(Number(obj.taxRate), 0, 100, 0),
   210→   210→    discountRate: clampNum(Number(obj.discountRate), 0, 100, 0),
   211→   211→    notes: typeof obj.notes === "string" ? obj.notes : "",
   212→   212→  };
   213→   213→}

   215→   215→function isValidTemplate(value: unknown): value is InvoiceTemplate {
   216→   216→  if (!value || typeof value !== "object") return false;
   217→   217→  const obj = value as Record<string, unknown>;
   218→   218→  return (
   219→   219→    typeof obj.id === "string" &&
   220→   220→    typeof obj.name === "string" &&
   221→   221→    Array.isArray(obj.lines) &&
   222→   222→    typeof obj.emisor === "object" &&
   223→   223→    typeof obj.cliente === "object" &&
   224→   224→    typeof obj.config === "object"
   225→   225→  );
   226→   226→}

   229→   229→function readTemplates(): InvoiceTemplate[] {
   230→   230→  if (typeof window === "undefined") return [];
   231→   231→  try {
   232→   232→    const raw = window.localStorage.getItem(TEMPLATES_KEY);
   233→   233→    if (!raw) return [];
   234→   234→    const parsed: unknown = JSON.parse(raw);
   235→   235→    if (!Array.isArray(parsed)) return [];
   236→   236→    return parsed.filter(isValidTemplate);
   237→   237→  } catch {
   238→   238→    return [];
   239→   239→  }

   322→   322→export default function InvoiceGeneratorTool() {
   323→   323→  const [emisor, setEmisor] = useState<InvoiceParty>(EMPTY_PARTY);
   324→   324→  const [cliente, setCliente] = useState<InvoiceParty>(EMPTY_PARTY);
   325→   325→  const [config, setConfig] = useState<InvoiceConfig>(defaultConfig);
   326→   326→  const [lines, setLines] = useState<InvoiceLine[]>(() => [defaultLine(0)]);
   327→   327→
   328→   328→  const [templates, setTemplates] = useState<InvoiceTemplate[]>(readTemplates);
   329→   329→  const [selectedTemplateId, setSelectedTemplateId] = useState("");
   330→   330→  const [saveDialogOpen, setSaveDialogOpen] = useState(false);
   331→   331→  const [templateName, setTemplateName] = useState("");
   332→   332→
   333→   333→  // Carga inicial de plantillas desde localStorage (ver readTemplates)
   334→   334→  const moneyFormatter = useMemo(
   335→   335→    () => new Intl.NumberFormat("es", { style: "currency", currency: config.currency }),
   336→   336→    [config.currency],
   337→   337→  );
   338→   338→
   339→   339→  const formatMoney = (value: number): string => moneyFormatter.format(numOr(value, 0));
   340→   340→
   341→   341→  const totals = useMemo<Totals>(() => {
   342→   342→    const items = lines.map((line) => {
   343→   343→      const amount = round2(numOr(line.quantity, 0) * numOr(line.unitPrice, 0));
   344→   344→      const tax = round2((amount * numOr(line.taxRate, 0)) / 100);
   345→   345→      return { amount, tax };
   346→   346→    });
   347→   347→    const subtotal = round2(items.reduce((acc, it) => acc + it.amount, 0));
   348→   348→    const discountRate = clampNum(config.discountRate, 0, 100, 0);
   349→   349→    const discount = round2((subtotal * discountRate) / 100);
   350→   350→    const factor = subtotal > 0 ? (subtotal - discount) / subtotal : 1;
   351→   351→    const taxes = round2(items.reduce((acc, it) => acc + it.tax, 0) * factor);
   352→   352→    const total = round2(subtotal - discount + taxes);
   353→   353→    return { items, subtotal, discount, taxes, total };
   354→   354→  }, [lines, config.discountRate]);
   355→   355→
   356→   356→  const updateEmisor = (patch: Partial<InvoiceParty>) =>
   357→   357→    setEmisor((prev) => ({ ...prev, ...patch }));
   358→   358→
   359→   359→  const updateCliente = (patch: Partial<InvoiceParty>) =>
   360→   360→    setCliente((prev) => ({ ...prev, ...patch }));
   361→   361→
   362→   362→  const updateConfig = (patch: Partial<InvoiceConfig>) =>
   363→   363→    setConfig((prev) => ({ ...prev, ...patch }));
   364→   364→
   365→   365→  const updateLine = (id: string, patch: Partial<InvoiceLine>) =>
   366→   366→    setLines((prev) =>
   367→   367→      prev.map((line) => (line.id === id ? { ...line, ...patch } : line)),
   368→   368→    );
   369→   369→
   370→   370→  const addLine = () =>
   371→   371→    setLines((prev) => [...prev, defaultLine(config.taxRate)]);
   372→   372→
   373→   373→  const removeLine = (id: string) => {
   374→   374→    setLines((prev) => {
   375→   375→      if (prev.length <= 1) return prev;
   376→   376→      return prev.filter((line) => line.id !== id);
   377→   377→    });
   378→   378→  };
   379→   379→

   381→   381→  const handleGlobalTaxChange = (rate: number) => {
   382→   382→    const previous = config.taxRate;
   383→   383→    updateConfig({ taxRate: rate });
   384→   384→    setLines((prev) =>
   385→   385→      prev.map((line) => (line.taxRate === previous ? { ...line, taxRate: rate } : line)),
   386→   386→    );
   387→   387→  };
   388→   388→

   390→   390→    setTemplates(next);
   391→   391→    try {
   392→   392→      window.localStorage.setItem(TEMPLATES_KEY, JSON.stringify(next));
   393→   393→    } catch {
   394→   394→      toast.error("No se pudo guardar la plantilla en este navegador");
   395→   395→    }
   396→   396→  };

   406→   406→    );

   409→   409→      name,
   410→   410→      savedAt: new Date().toISOString(),
   411→   411→      emisor,
   412→   412→      cliente,
   413→   413→      config,
   414→   414→      lines,
   415→   415→    };
   416→   416→    const next = existing
   417→   417→      ? templates.map((t) => (t.id === existing.id ? template : t))
   418→   418→      : [...templates, template];
   419→   419→    persistTemplates(next);
   420→   420→    setSelectedTemplateId(template.id);
   421→   421→    setSaveDialogOpen(false);
   422→   422→    setTemplateName("");
   423→   423→    toast.success(existing ? `Plantilla "${name}" actualizada` : `Plantilla "${name}" guardada`);
   424→   424→  };
   425→   425→
   426→   426→  const loadTemplate = () => {
   427→   427→    const template = templates.find((t) => t.id === selectedTemplateId);
   428→   428→    if (!template) {
   429→   429→      toast.error("Selecciona una plantilla para cargar");
   430→   430→      return;
   431→   431→    }
   432→   432→    setEmisor(normalizeParty(template.emisor));
   433→   433→    setCliente(normalizeParty(template.cliente));
   434→   434→    setConfig(normalizeConfig(template.config));
   435→   435→    setLines(
   436→   436→      template.lines.length > 0 ? template.lines.map(normalizeLine) : [defaultLine(0)],
   437→   437→    );
   438→   438→    setSelectedTemplateId("");
   439→   439→    toast.success(`Plantilla "${template.name}" cargada`);
   440→   440→  };
   441→   441→
   442→   442→  const deleteTemplate = () => {
   443→   443→    const template = templates.find((t) => t.id === selectedTemplateId);
   444→   444→    if (!template) {
   445→   445→      toast.error("Selecciona la plantilla que quieres eliminar");
   446→   446→      return;
   447→   447→    }
   448→   448→    persistTemplates(templates.filter((t) => t.id !== template.id));
   449→   449→    setSelectedTemplateId("");
   450→   450→    toast.success(`Plantilla "${template.name}" eliminada`);
   451→   451→  };
   452→   452→
   453→   453→  const resetInvoice = () => {

   458→   458→    setSelectedTemplateId("");
   459→   459→    toast.success("Se creó una factura en blanco");
   460→   460→  };
   461→   461→
   462→   462→  const newInvoice = () => {
   463→   463→    toast("Se creará una factura en blanco. Los datos no guardados se perderán.", {
   464→   464→      action: { label: "Confirmar", onClick: resetInvoice },
   465→   465→      duration: 8000,
   466→   466→    });
   467→   467→  };
   468→   468→
   469→   469→  const validateForPdf = (): string[] => {
   470→   470→    const problems: string[] = [];
   471→   471→    if (!emisor.name.trim()) problems.push("el nombre del emisor");
   472→   472→    if (!cliente.name.trim()) problems.push("el nombre del cliente");
   473→   473→    if (!lines.some((line) => line.description.trim())) {
   474→   474→      problems.push("al menos una línea con descripción");
   475→   475→    }
   476→   476→    return problems;
   477→   477→  };
   478→   478→
   479→   479→  const exportPdf = () => {
   480→   480→    const problems = validateForPdf();
   481→   481→    if (problems.length > 0) {

   891→   891→/* ---------- Generación del PDF (jsPDF + autoTable) ---------- */
   892→   892→
   893→   893→interface PdfPayload {
   894→   894→  emisor: InvoiceParty;
   895→   895→  cliente: InvoiceParty;
   896→   896→  config: InvoiceConfig;
   897→   897→  lines: InvoiceLine[];
   898→   898→  totals: Totals;
   899→   899→  formatMoney: (value: number) => string;
   900→   900→}
   901→   901→
   902→   902→function buildInvoicePdf(payload: PdfPayload): void {
   903→   903→  const { emisor, cliente, config, lines, totals, formatMoney } = payload;
   904→   904→
   905→   905→  const doc = new jsPDF({ orientation: "portrait", unit: "mm", format: "a4" });
   906→   906→  const docWithTable = doc as unknown as { lastAutoTable?: { finalY: number } };
   907→   907→
   908→   908→  const pageWidth = doc.internal.pageSize.getWidth();
   909→   909→  const pageHeight = doc.internal.pageSize.getHeight();
   910→   910→  const margin = 15;
   911→   911→  const columnWidth = (pageWidth - margin * 2 - 10) / 2;
   912→   912→
   913→   913→  // Encabezado
   914→   914→  doc.setFont("helvetica", "bold");
   915→   915→  doc.setFontSize(24);
   916→   916→  doc.setTextColor(20, 20, 20);
   917→   917→  doc.text("FACTURA", margin, 24);
   918→   918→  doc.setFontSize(13);
   919→   919→  doc.setTextColor(100, 100, 100);
   920→   920→  doc.text(`N.º ${config.number.trim() || "FAC-001"}`, margin, 31);
   921→   921→
   922→   922→  doc.setFont("helvetica", "normal");
   923→   923→  doc.setFontSize(9);
   924→   924→  doc.setTextColor(100, 100, 100);
   925→   925→  const rightX = pageWidth - margin;
   926→   926→  const issue = parseDateInput(config.issueDate);
   927→   927→  const due = parseDateInput(config.dueDate);
   928→   928→  let headerY = 22;
   929→   929→  doc.text(`Fecha de emisión: ${issue ? formatDate(issue) : "—"}`, rightX, headerY, {
   930→   930→    align: "right",
   931→   931→  });
   932→   932→  headerY += 5;
